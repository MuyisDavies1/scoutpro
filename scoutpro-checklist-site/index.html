Platform Setup Commands

Run these from inside scoutpro-capacitor/

# 1. One-time setup (installs deps + adds platforms)
cd scoutpro-capacitor
bash scripts/setup.sh

# 2. After any code change, sync web assets to native layers
npx cap sync

# 3a. Open in Xcode (macOS only)
npx cap open ios

# 3b. Open in Android Studio (any OS)
npx cap open android

# 4. Run on a connected device / emulator
npx cap run ios        --target "iPhone 15 Pro"
npx cap run android   --target emulator-5554
# macOS + Xcode 15+ required
xcode-select --install          # install CLI tools if missing
sudo gem install cocoapods         # or: brew install cocoapods

# Add iOS platform (first time only)
npx cap add ios

# Sync + open Xcode
npx cap sync
npx cap open ios

# In Xcode:
#  1. Signing & Capabilities → select your Apple Developer Team
#  2. Bundle Identifier → com.scoutpro.app
#  3. Deployment Target → iOS 14.0
#  4. Product → Archive → Distribute App → App Store Connect

# Build for release (after Xcode setup)
npx cap build ios --configuration release
# Android Studio + JDK 17 required (all platforms)
# Download: https://developer.android.com/studio

# Add Android platform (first time only)
npx cap add android

# Sync + open Android Studio
npx cap sync
npx cap open android

# In Android Studio:
#  1. Build → Generate Signed Bundle/APK
#  2. Choose: Android App Bundle (.aab) — required for Play Store
#  3. Create or select your .jks keystore
#  4. Build Variant: Release

# Or build from command line
cd android
./gradlew bundleRelease
# Output: app/build/outputs/bundle/release/app-release.aab
══ iOS Signing ══════════════════════════════════════════════
# 1. Enroll in Apple Developer Program ($99/year)
#    https://developer.apple.com/programs/

# 2. Create App ID in Developer portal:
#    Identifiers → + → App IDs → com.scoutpro.app
#    Enable: Push Notifications, Sign in with Apple (optional)

# 3. Create distribution certificate in Xcode:
#    Preferences → Accounts → Manage Certificates → +
#    → Apple Distribution

# 4. Create App Store provisioning profile:
#    Developer portal → Profiles → Distribution → App Store

══ Android Signing ══════════════════════════════════════════
# Generate release keystore (one time, keep this safe!)
keytool -genkey -v \
  -keystore scoutpro-release.jks \
  -alias scoutpro \
  -keyalg RSA -keysize 2048 \
  -validity 10000

# Set credentials in gradle.properties (NOT in build.gradle)
KEYSTORE_PATH=../scoutpro-release.jks
KEYSTORE_PASSWORD=your_secure_password
KEY_ALIAS=scoutpro
KEY_PASSWORD=your_key_password

# NEVER commit .jks or gradle.properties to git!
# Add to .gitignore: *.jks, gradle.properties
